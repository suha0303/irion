cmake_minimum_required(VERSION 3.22)
project(drone_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(drone_powertrain STATIC physics/powertrain.cpp)
set_target_properties(drone_powertrain PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(drone_powertrain PUBLIC physics)
target_compile_options(drone_powertrain PRIVATE -Wall -Wextra -Werror)
set(DRONE_PROFILE_HEADER "" CACHE FILEPATH "Explicit vehicle profile header")
if(DRONE_PROFILE_HEADER)
  target_compile_definitions(drone_powertrain PUBLIC DRONE_PROFILE_HEADER="${DRONE_PROFILE_HEADER}")
  add_executable(collection_numeric physics/test_collection_numeric.cpp)
  target_link_libraries(collection_numeric PRIVATE drone_powertrain)
endif()
add_executable(powertrain_test physics/test_powertrain.cpp)
target_link_libraries(powertrain_test PRIVATE drone_powertrain)
add_executable(profile_physics_test physics/test_multirotor.cpp)
target_link_libraries(profile_physics_test PRIVATE drone_powertrain)
enable_testing()
add_test(NAME powertrain COMMAND powertrain_test)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
add_library(DronePowertrain SHARED adapters/PowertrainSystem.cpp)
target_link_libraries(DronePowertrain PRIVATE drone_powertrain gz-sim8::gz-sim8 gz-transport13::core)
target_compile_options(DronePowertrain PRIVATE -Wall -Wextra -Werror)
add_executable(powertrain_bench physics/bench.cpp)
target_link_libraries(powertrain_bench PRIVATE drone_powertrain)
add_executable(rc_bench physics/rc_bench.cpp)
target_link_libraries(rc_bench PRIVATE drone_powertrain)

add_executable(boundaries_test physics/test_boundaries.cpp)
target_link_libraries(boundaries_test PRIVATE drone_powertrain)
add_test(NAME boundaries COMMAND boundaries_test)
add_test(NAME profile_physics COMMAND profile_physics_test)

add_executable(capture_timing_fixture tools/capture_timing_fixture.cpp)
target_compile_options(capture_timing_fixture PRIVATE -Wall -Wextra -Werror)
